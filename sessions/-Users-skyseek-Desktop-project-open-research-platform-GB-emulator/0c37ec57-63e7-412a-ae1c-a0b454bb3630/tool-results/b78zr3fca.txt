    28	pub struct Machine {
    29	    cpu: Cpu,
    30	    bus: Bus,
    31	    /// `run_frame` 的 T-cycle 余量：每帧加 70224，执行到不大于 0；超出部分累计到下一帧。
    32	    frame_budget: i64,
    33	}
    34	
    35	impl Machine {
    36	    /// 解析头部、选择 MBC、设置 DMG 开机状态。
    37	    pub fn new(rom: Vec<u8>) -> Result<Self, CartError> {
    38	        if rom.len() < 0x150 {
    39	            return Err(CartError::BadSize(rom.len()));
    40	        }
    41	        let header_checksum = rom[0x14D];
    42	        let cart = cart::from_rom(rom)?;
    43	        let mut cpu = Cpu::new();
    44	        // Power_Up_Sequence：头部校验和为 0 时 H、C 清零，否则置位（Z=1、N=0 不变）。
    45	        cpu.f = if header_checksum == 0 { 0x80 } else { 0xB0 };
    46	        Ok(Self { cpu, bus: Bus::new(cart), frame_budget: 0 })
    47	    }
    48	
    49	    /// 执行 1 条指令或 1 次中断分派，返回消耗的 T 数；HALT/STOP/锁死时也至少返回 4，绝不返回 0。
    50	    pub fn step(&mut self) -> u32 {
    51	        let start = self.bus.mcycles();
    52	        self.cpu.step(&mut self.bus);
    53	        ((self.bus.mcycles() - start) * 4) as u32
    54	    }
    55	
    56	    /// 精确推进 70224 T（余量累计到下一帧），与 LCD 开关无关。
    57	    pub fn run_frame(&mut self) {
    58	        self.frame_budget += FRAME_T as i64;
    59	        while self.frame_budget > 0 {
    60	            self.frame_budget -= self.step() as i64;
    61	        }
    62	    }
    63	
    64	    /// bit0..7 = A B Select Start Right Left Up Down，1 = 按下（见 `crate::buttons`）。
    65	    pub fn set_buttons(&mut self, mask: u8) {
    66	        self.bus.set_buttons(mask);
    67	    }
    68	
    69	    /// 最近一个完整帧（前缓冲），值为 0xFF/0xAA/0x55/0x00。
    70	    pub fn frame(&self) -> &Frame {
    71	        self.bus.frame()
    72	    }
    73	
    74	    /// 无副作用读（不推进时钟，不触发任何寄存器副作用）。
    75	    pub fn peek(&self, addr: u16) -> u8 {
    76	        self.bus.peek(addr)
    77	    }
    78	
    79	    /// 取走自上次调用以来串口发出的字节（内部时钟传输开始时追加）。
    80	    pub fn take_serial(&mut self) -> Vec<u8> {
    81	        self.bus.take_serial()
    82	    }
    83	
    84	    pub fn regs(&self) -> Regs {
    85	        let c = &self.cpu;
    86	        Regs {
    87	            a: c.a,
    88	            f: c.f,
    89	            b: c.b,
    90	            c: c.c,
    91	            d: c.d,
    92	            e: c.e,
    93	            h: c.h,
    94	            l: c.l,
    95	            sp: c.sp,
    96	            pc: c.pc,
    97	            ime: c.ime,
    98	            halted: c.halted,
    99	        }
   100	    }
   101	
   102	    /// 自上次查询以来是否执行过 0x40（LD B,B），读后清零。
   103	    pub fn ld_b_b_hit(&mut self) -> bool {
   104	        std::mem::take(&mut self.cpu.ld_b_b)
   105	    }
   106	
   107	    /// 开机以来累计的 T-cycle 数。
   108	    pub fn cycles(&self) -> u64 {
   109	        self.bus.mcycles() * 4
   110	    }
   111

     8	/// CPU 看到的总线。每次调用恰好消耗 1 个 M-cycle（`pending`/`ack` 除外，它们不耗时）。
     9	pub trait CpuBus {
    10	    /// 读一个字节：1 M-cycle。
    11	    fn read(&mut self, addr: u16) -> u8;
    12	    /// 写一个字节：1 M-cycle。
    13	    fn write(&mut self, addr: u16, v: u8);
    14	    /// 内部周期：1 M-cycle，无总线访问。
    15	    fn idle(&mut self);
    16	    /// `IE & IF & 0x1F`，不耗时。
    17	    fn pending(&self) -> u8;
    18	    /// 清除 IF 的第 `bit` 位（`bit` 为序号 0..=4，0 = VBlank），不耗时。
    19	    fn ack(&mut self, bit: u8);
    20	    /// 执行 STOP 时调用（不耗时）：Bus 借此清零 DIV。测试用的假总线可以不实现。
    21	    fn stop(&mut self) {}
    22	}

    24	/// 寄存器与 CPU 状态。`Cpu::new` 给出 DMG 开机（boot ROM 结束、PC=0x0100）时的值；

    34	    pub e: u8,

    39	    pub ime: bool,
    40	    pub halted: bool,
    41	    /// 执行 0x40（LD B,B）时置位；`Machine::ld_b_b_hit` 读取后清零。

    92	#[derive(Debug)]
    93	pub struct StateReader<'a> {
    94	    data: &'a [u8],
    95	    pos: usize,
    96	}
    97	
    98	impl<'a> StateReader<'a> {
    99	    pub fn new(data: &'a [u8]) -> Self {
   100	        Self { data, pos: 0 }
   101	    }
   102	    pub fn remaining(&self) -> usize {
   103	        self.data.len() - self.pos
   104	    }
   105	    fn take(&mut self, n: usize) -> Result<&'a [u8], StateError> {
   106	        if self.remaining() < n {
   107	            return Err(StateError::Truncated);
   108	        }
   109	        let s = &self.data[self.pos..self.pos + n];
   110	        self.pos += n;
   111	        Ok(s)
   112	    }
   113	    pub fn tag(&mut self, t: &[u8; 4]) -> Result<(), StateError> {
   114	        if self.take(4)? == t {
   115	            Ok(())
   116	        } else {
   117	            Err(StateError::Corrupt("module tag mismatch"))
   118	        }
   119	    }
   120	    pub fn u8(&mut self) -> Result<u8, StateError> {
   121	        Ok(self.take(1)?[0])
   122	    }
   123	    pub fn bool(&mut self) -> Result<bool, StateError> {
   124	        match self.u8()? {
   125	            0 => Ok(false),
   126	            1 => Ok(true),
   127	            _ => Err(StateError::Corrupt("bool")),
   128	        }
   129	    }
   130	    pub fn u16(&mut self) -> Result<u16, StateError> {
   131	        Ok(u16::from_le_bytes(self.take(2)?.try_into().unwrap()))
   132	    }
   133	    pub fn u32(&mut self) -> Result<u32, StateError> {
   134	        Ok(u32::from_le_bytes(self.take(4)?.try_into().unwrap()))
   135	    }
   136	    pub fn u64(&mut self) -> Result<u64, StateError> {
   137	        Ok(u64::from_le_bytes(self.take(8)?.try_into().unwrap()))
   138	    }
   139	    pub fn i64(&mut self) -> Result<i64, StateError> {
   140	        Ok(i64::from_le_bytes(self.take(8)?.try_into().unwrap()))
   141	    }
   142	    pub fn bytes_into(&mut self, out: &mut [u8]) -> Result<(), StateError> {
   143	        out.copy_from_slice(self.take(out.len())?);
   144	        Ok(())
   145	    }
   146	    pub fn vec(&mut self) -> Result<Vec<u8>, StateError> {
   147	        let n = self.u32()? as usize;
   148	        Ok(self.take(n)?.to_vec())
   149	    }
   150	}

     3	//! 冻结部分（D001）：`Ppu::new`、VRAM/OAM 访问方法、`dma_write_oam`、`frame`、`Peripheral` 实现
     4	//! （`tick_mcycle` 返回 VBlank/STAT 中断位；`read_reg`/`write_reg` 处理 FF40–FF45、FF47–FF4B；

    32	    pub fn new() -> Self {
    33	        let mut regs = [0u8; 12];

    94	        }
    95	    }

    48	pub fn from_rom(rom: Vec<u8>) -> Result<Box<dyn Cart>, CartError> {
    49	    if rom.len() < 0x150 {
